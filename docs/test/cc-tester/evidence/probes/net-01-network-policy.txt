// Permit only the packaged Runtime control-plane process to create/stop
// per-Worker transient scopes.  A scope inherits the caller's already-dropped
// identity and cannot carry User=/Group= properties.  Service units are not
// allowed here: accepting them would let a compromised Worker request a
// privileged transient service by adding a User= override to a valid unit.
function bianbuRuntimeOwner(unit) {
    var match = /^bianbu-agent-runtime@([A-Za-z_][A-Za-z0-9_.-]*)\.service$/.exec(unit || "");
    return match ? match[1] : "";
}

polkit.addRule(function(action, subject) {
    if (action.id !== "org.freedesktop.systemd1.manage-units") {
        return polkit.Result.NOT_HANDLED;
    }
    var unit = action.lookup("unit") || "";
    var verb = action.lookup("verb") || "";
    var requestedOwner = bianbuRuntimeOwner(unit);
    if (requestedOwner && requestedOwner === subject.user &&
        (verb === "start" || verb === "stop" || verb === "restart")) {
        return polkit.Result.YES;
    }
    var runtimeOwner = bianbuRuntimeOwner(subject.system_unit);
    if (!subject.no_new_privileges || !runtimeOwner ||
        runtimeOwner !== subject.user) {
        return polkit.Result.NOT_HANDLED;
    }
    if (!/^bianbu-agent-[A-Za-z0-9_.-]+\.scope$/.test(unit)) {
        return polkit.Result.NOT_HANDLED;
    }
    if (verb === "start" || verb === "stop" || verb === "kill" ||
        verb === "restart" || verb === "reload" || verb === "reset-failed") {
        return polkit.Result.YES;
    }
    return polkit.Result.NOT_HANDLED;
});
